cmake_minimum_required(VERSION 3.21)
if(NOT DEFINED CMAKE_HIP_ARCHITECTURES)
  set(CMAKE_HIP_ARCHITECTURES gfx1100)
endif()
project(bmms_tiled LANGUAGES CXX HIP)
add_library(bmms_tiled SHARED src/tiled.hip)
target_include_directories(bmms_tiled PUBLIC include PRIVATE ../include)
set_target_properties(bmms_tiled PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
target_compile_options(bmms_tiled PRIVATE -Wall -Wextra -Werror)
