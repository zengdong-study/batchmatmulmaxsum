cmake_minimum_required(VERSION 3.21)
if(NOT DEFINED CMAKE_HIP_ARCHITECTURES)
  set(CMAKE_HIP_ARCHITECTURES gfx1100)
endif()
project(bmms_split_mn LANGUAGES CXX HIP)
add_library(bmms_split_mn SHARED src/split_mn.hip)
target_include_directories(bmms_split_mn PUBLIC include ../include)
set_target_properties(bmms_split_mn PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
target_compile_options(bmms_split_mn PRIVATE -Wall -Wextra -Werror)
