cmake_minimum_required(VERSION 3.21)
if(NOT DEFINED CMAKE_HIP_ARCHITECTURES)
  set(CMAKE_HIP_ARCHITECTURES gfx1100)
endif()
project(bmms_tiled LANGUAGES CXX HIP)
add_library(bmms_tiled SHARED src/tiled.hip)
target_include_directories(bmms_tiled PUBLIC include PRIVATE ../include)
set_target_properties(bmms_tiled PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
target_compile_options(bmms_tiled PRIVATE -Wall -Wextra -Werror)
option(BMMS_BUILD_S3 "Build the fixed host S3 entry" OFF)
set(BMMS_S1_BUILD /tmp/bmms-s1-build CACHE PATH "Frozen S1 build directory")
if(BMMS_BUILD_S3)
  add_library(bmms_s3 SHARED src/s3_entry.hip)
  target_include_directories(bmms_s3 PUBLIC include PRIVATE ../include)
  target_link_libraries(bmms_s3 PRIVATE bmms_tiled ${BMMS_S1_BUILD}/libbmms_s1.so)
  set_target_properties(bmms_s3 PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
  target_compile_options(bmms_s3 PRIVATE -Wall -Wextra -Werror)
endif()

option(BMMS_BUILD_S4 "Build the fixed host S4 entry" OFF)
if(BMMS_BUILD_S4)
  add_library(bmms_s4 SHARED src/s4_entry.hip)
  target_include_directories(bmms_s4 PUBLIC include PRIVATE ../include)
  target_link_libraries(bmms_s4 PRIVATE bmms_tiled ${BMMS_S1_BUILD}/libbmms_s1.so)
  set_target_properties(bmms_s4 PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
  target_compile_options(bmms_s4 PRIVATE -Wall -Wextra -Werror)
endif()
